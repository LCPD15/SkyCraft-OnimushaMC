cmake_minimum_required(VERSION 3.24)
project(OnimushaMC VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OnimushaMC requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(omc_runtime_smoke_host native/RuntimeSmokeHost.cpp)
target_link_libraries(omc_runtime_smoke_host PRIVATE omc_bridge)
add_executable(omc_restart_smoke_host native/RestartSmokeHost.cpp)
target_link_libraries(omc_restart_smoke_host PRIVATE omc_bridge)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "native/vendor/lua/*.c")
add_library(omc_lua STATIC ${LUA_SOURCES})
target_include_directories(omc_lua PUBLIC native/vendor/lua)
target_compile_definitions(omc_lua PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(omc_bridge STATIC native/Bridge.cpp native/RenderStore.cpp native/CollisionPacking.cpp)
target_compile_definitions(omc_bridge PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(omc_bridge PUBLIC advapi32 user32)
option(OMC_BUILD_PLUGIN "Build game-loadable REFramework plugin" ON)
add_library(omc_plugin_api OBJECT native/Plugin.cpp)
target_include_directories(omc_plugin_api PRIVATE native/vendor)
target_compile_definitions(omc_plugin_api PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(OMC_BUILD_PLUGIN)
    add_library(OnimushaMC SHARED $<TARGET_OBJECTS:omc_plugin_api> native/Renderer.cpp)
    target_link_libraries(OnimushaMC PRIVATE omc_bridge omc_lua d3d11 d3d12 dxgi d3dcompiler)
    target_compile_definitions(OnimushaMC PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(OnimushaMC PRIVATE native/vendor)
    set_target_properties(OnimushaMC PROPERTIES OUTPUT_NAME OnimushaMC)
endif()
if(MSVC)
    target_compile_options(omc_bridge PRIVATE /W4 /permissive- /EHsc)
    target_compile_options(omc_plugin_api PRIVATE /W4 /permissive- /EHsc)
    if(OMC_BUILD_PLUGIN)
        target_compile_options(OnimushaMC PRIVATE /W4 /permissive- /EHsc)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(omc_native_tests native/BridgeTests.cpp)
    target_link_libraries(omc_native_tests PRIVATE omc_bridge)
    add_test(NAME native_mapping_and_decoder COMMAND omc_native_tests)
    if(OMC_BUILD_PLUGIN)
        add_executable(omc_api_tests native/ApiTests.cpp)
        target_include_directories(omc_api_tests PRIVATE native/vendor)
        target_link_libraries(omc_api_tests PRIVATE omc_lua d3d12 dxgi user32)
        target_compile_definitions(omc_api_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_dependencies(omc_api_tests OnimushaMC)
        add_test(NAME native_lua_api COMMAND omc_api_tests $<TARGET_FILE:OnimushaMC>)
        set_tests_properties(native_lua_api PROPERTIES RUN_SERIAL TRUE SKIP_RETURN_CODE 77)
        add_executable(omc_renderer_tests native/RendererSmokeTests.cpp native/Renderer.cpp)
        target_include_directories(omc_renderer_tests PRIVATE native/vendor)
        target_compile_definitions(omc_renderer_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(omc_renderer_tests PRIVATE omc_bridge d3d11 d3d12 d3dcompiler dxgi)
        add_test(NAME native_d3d12_renderer COMMAND omc_renderer_tests)
        set_tests_properties(native_d3d12_renderer PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    endif()
endif()
